================================================================================
Space between an attribute and its arguments
:error
================================================================================

type A = { @deprecated ("x") x: Int }

--------------------------------------------------------------------------------


================================================================================
Attribute on a spread
:error
================================================================================

type A = { @deprecated ...B }

--------------------------------------------------------------------------------


================================================================================
Row parameter
:error
================================================================================

type A r = { x: Int, ...r }

--------------------------------------------------------------------------------


================================================================================
Space after a spread
:error
================================================================================

type A = < ... B >

--------------------------------------------------------------------------------


================================================================================
Parameter starting with an underscore
:error
================================================================================

type P _x = { a: _x }

--------------------------------------------------------------------------------


================================================================================
Import without names
:error
================================================================================

import std/time ()

--------------------------------------------------------------------------------


================================================================================
Import of a full version
:error
================================================================================

import @haus-fmt/money@1.1.0 (Money)

--------------------------------------------------------------------------------


================================================================================
Import of a major version with a leading zero
:error
================================================================================

import @haus-fmt/money@01 (Money)

--------------------------------------------------------------------------------


================================================================================
Import of an empty version
:error
================================================================================

import @haus-fmt/money@ (Money)

--------------------------------------------------------------------------------


================================================================================
Space before an optional marker
:error
================================================================================

type A = { a ?: Int }

--------------------------------------------------------------------------------


================================================================================
Trailing bar in a variant
:error
================================================================================

type A = < A | >

--------------------------------------------------------------------------------


================================================================================
Lowercase type name
:error
================================================================================

type a = Int

--------------------------------------------------------------------------------
